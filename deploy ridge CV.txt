1. apply_export : même calcul de features que deploy.predict, et vérification de la normalisation. Remplace la fonction entière par :

def apply_export(export, rows, norm_config=None):
    """
    Apply an exported ridge to new records, labelled or not.

    Features are built with deploy.features_for_inference, the exact path used
    by deploy.predict, so the exported model scores rows identically to the
    bundle it was taken from.

    Args:
        export: Output of ridge_export.
        rows: Records to score.
        norm_config: Normalization used for these records, checked against the
            one recorded at training time.

    Returns:
        Predicted WER per segment.
    """
    from deploy import entries_to_matrix, features_for_inference

    if norm_config is not None and export.get("norm_config") and \
            dict(norm_config) != export["norm_config"]:
        raise ValueError(f"normalization mismatch: trained with "
                         f"{export['norm_config']}, called with {dict(norm_config)}")

    entries = features_for_inference(rows, tuple(export["blocks"]),
                                     tuple(export["roles"]))
    seen = {key for entry in entries for key in entry}
    missing = [n for n in export["feature_names"] if n not in seen]
    if missing:
        raise ValueError(f"features missing from the records: {missing}")

    X = entries_to_matrix(entries, export["feature_names"])
    standardized = (X - np.array(export["scaler_mean"])) \
        / np.array(export["scaler_scale"])
    raw = standardized @ np.array(export["coef"]) + export["intercept"]
    return np.clip(raw, *export["clip"])
